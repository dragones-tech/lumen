messaging · independent views, one event

A catalog, a cart badge in the header, and toasts — three independent views. The badge subscribes to the cart Collection's structural events; the toasts subscribe to an explicit app bus. Neither knows about the catalog. Every subscription is bound to this.signal, so it cleans up on unmount — no leaks, no global Radio.

Code

import { View, Model, Collection, CollectionView, EventEmitter } from '../../src/index.js';

// Two explicit, shared channels — imported where needed. No global Radio, no magic.
const bus  = new EventEmitter();          // app-wide notifications
const cart = new Collection([], Model);   // shared cart; emits add/remove/reset

class Product extends View {
  static template = '#product';
  onMount() {
    this.ui.name.textContent  = this.props.model.get('name');
    this.ui.price.textContent = '$' + this.props.model.get('price');
    this.listen(this.ui.add, 'click', this.add);
  }
  add = () => {
    cart.add(this.props.model);                                   // structural event → badge reacts
    bus.emit('notify', { text: `Added ${this.props.model.get('name')}` }); // app bus → toasts react
  };
}

class CartBadge extends View {
  static template = '#bar';
  onMount() {
    this.update();
    // Subscribe straight to the Collection — no coupling to the catalog. Auto-cleaned via signal.
    cart.on('add',    this.update, { signal: this.signal });
    cart.on('remove', this.update, { signal: this.signal });
    cart.on('reset',  this.update, { signal: this.signal });
    this.listen(this.ui.clear, 'click', () => { cart.reset(); bus.emit('notify', { text: 'Cart cleared' }); });
  }
  update = () => { this.ui.count.textContent = String(cart.length); };
}

class Toasts extends View {
  static template = '#toasts';
  onMount() { bus.on('notify', this.show, { signal: this.signal }); }   // listens on the app bus
  show = ({ text }) => {
    const li = Object.assign(document.createElement('li'), { className: 'toast', textContent: text });
    this.el.appendChild(li);
    setTimeout(() => li.remove(), 2000);
  };
}

class Catalog extends CollectionView {        // one Product view per model, kept in sync
  static template = '#catalog'; static childView = Product; static container = 'list';
}

// Three independent views — they coordinate ONLY through the shared `cart` and `bus`.
new CartBadge().mount(document.querySelector('#header'));
new Catalog({ collection: new Collection(products, Model) }).mount(document.querySelector('#shop'));
new Toasts().mount(document.querySelector('#notices'));
