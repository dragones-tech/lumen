Http · fetch + Collection + CollectionView

Loads users from a public API into a Collection, rendered by a CollectionView. The request uses this.signal, so it cancels on unmount. timeout 1ms forces a TimeoutError via the timeout option (built on AbortSignal.timeout + AbortSignal.any). (Needs internet.)

Code

import { Http, View, Model, Collection, CollectionView } from '../../src/index.js';

const api = new Http({ baseURL: 'https://jsonplaceholder.typicode.com' });

class UserItem extends View {
  static template = '#user-item';
  onMount() {
    this.ui.name.textContent  = this.props.model.get('name');
    this.ui.email.textContent = this.props.model.get('email');
  }
}

class UserList extends CollectionView {
  static template  = '#user-list';
  static childView = UserItem;
  static container = 'list';
  onMount() {
    super.onMount();
    this.listen(this.ui.reload, 'click', () => this.load());
    this.listen(this.ui.slow, 'click', () => this.load({ timeout: 1 }));
    this.load();
  }
  load = async (opts = {}) => {
    this.ui.status.textContent = 'Loading…';
    try {
      const data = await api.get('/users', { ...opts, signal: this.signal });
      this.collection.reset(data);   // CollectionView reconciles
      this.ui.status.textContent = data.length + ' users';
    } catch (e) {
      if (e.name === 'TimeoutError') this.ui.status.textContent = 'Timed out ⏱';
      else if (e.name !== 'AbortError') this.ui.status.textContent = 'Error: ' + e.message;
    }
  };
}

new UserList({ collection: new Collection([], Model) }).mount(app);
