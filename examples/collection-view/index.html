CollectionView · keyed reconciliation

Each todo is a Model; the list is a Collection; the UI is a CollectionView. Add/remove touch only one child view (slide in/out). The footer reacts to model changes. Nothing else re-renders.

Code

import { Collection, Model, View, CollectionView, slideIn, slideOut } from '../../src/index.js';

class Todo extends Model {}

class TodoItem extends View {
  static template = '#todo-item';
  onMount() {
    const { model } = this.props;
    this.ui.text.textContent = model.get('text');
    this.ui.check.checked = model.get('done');
    this.paint();
    this.listen(this.ui.check, 'change', this.toggle);
    this.listen(this.ui.remove, 'click', this.remove);
    model.on('change:done', this.paint, { signal: this.signal });
  }
  toggle = () => { this.props.model.set('done', this.ui.check.checked); this.props.onToggle(); };
  remove = () => this.props.onRemove();
  paint = () => this.el.classList.toggle('done', this.props.model.get('done'));
  animateIn()  { return slideIn(this.el); }
  animateOut() { return slideOut(this.el); }
}

class TodoList extends CollectionView {
  static template  = '#todo-list';
  static childView = TodoItem;
  static container = 'items';          // children mount into ui.items
  onMount() {
    super.onMount();                       // sets up keyed reconciliation
    this.listen(this.ui.form, 'submit', this.onSubmit);
    this.collection.on('add', this.refreshCount, { signal: this.signal });
    this.collection.on('remove', this.refreshCount, { signal: this.signal });
  }
  onSubmit = (e) => {
    e.preventDefault();
    const text = this.ui.input.value.trim();
    if (text) this.collection.add({ text, done: false });
    this.ui.input.value = ''; this.ui.input.focus();
  };
  refreshCount = () => { this.ui.count.textContent = this.collection.where({ done: false }).length; };
  childProps(model) {
    return {
      onRemove: () => this.collection.remove(model),
      onToggle: () => this.refreshCount(),
    };
  }
}

const todos = new Collection([{ text: 'try Lumen', done: false }], Todo);
new TodoList({ collection: todos }).mount(app);
