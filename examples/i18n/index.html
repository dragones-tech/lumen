I18n · reactive translations

Switch the locale and the view re-renders its text (subscribed via this.signal). Shows interpolation and nested keys.

Code

import { I18n, View } from '../../src/index.js';

const i18n = new I18n({ locale: 'en', messages: {
  en: { title: 'Welcome', greeting: 'Hello, {name}!', items: '{n} items' },
  es: { title: 'Bienvenido', greeting: '¡Hola, {name}!', items: '{n} elementos' },
}});

class Panel extends View {
  static template = '#panel';
  onMount() {
    this.paint();
    this.listen(this.ui.name, 'input', this.paint);
    this.listen(this.ui.en, 'click', () => i18n.setLocale('en'));
    this.listen(this.ui.es, 'click', () => i18n.setLocale('es'));
    i18n.onChange(this.paint, { signal: this.signal }); // re-render on locale change
  }
  paint = () => {
    this.ui.title.textContent    = i18n.t('title');
    this.ui.greeting.textContent = i18n.t('greeting', { name: this.ui.name.value || '?' });
    this.ui.count.textContent    = i18n.t('items', { n: 3 });
  };
}

new Panel().mount(app);
