View · classes, lifecycle, animation

A Board view manages Note child views. Notes slide in; on remove they slide out before leaving the DOM. The input keeps focus — nothing is re-rendered. Check the console for cleanup logs.

Code

import { View, slideIn, slideOut } from '../../src/index.js';

class Note extends View {
  static template = '#note';
  onMount() {
    this.ui.text.textContent = this.props.text;
    this.listen(this.ui.remove, 'click', () => this.props.onRemove(this));
  }
  onUnmount() { console.log('Note cleaned up:', this.props.text); }
  animateIn()  { return slideIn(this.el); }
  animateOut() { return slideOut(this.el); } // finishes before removal
}

class Board extends View {
  static template = '#board';
  onMount() {
    this.listen(this.ui.form, 'submit', (e) => { e.preventDefault(); this.add(); });
    this.listen(this.ui.clear, 'click', () => this.clearAll());
  }
  add() {
    const text = this.ui.input.value.trim();
    if (!text) return;
    this.ui.input.value = '';
    this.ui.input.focus();          // focus kept — no re-render
    this.addChild(new Note({ text, onRemove: (n) => this.removeChild(n) }), this.ui.list);
  }
  async clearAll() {
    for (const n of [...this.children]) await this.removeChild(n);
  }
}

new Board().mount(app);
