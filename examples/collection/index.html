Collection · events + non-mutating queries

The ordered list is the source of truth. where and sort return results but never reorder it — watch the list stay put while the log shows the query result.

Code

import { Collection, Model, View } from '../../src/index.js';

class Todo extends Model {}

class Demo extends View {
  static template = '#demo';
  onCreate() {
    this.todos = new Collection([{ id: 1, text: 'b', done: false }], Todo);
    this.nextId = 2;
  }
  onMount() {
    // re-render the list whenever the collection structurally changes
    for (const e of ['add', 'remove', 'reset'])
      this.todos.on(e, this.renderList, { signal: this.signal });
    this.listen(this.ui.add, 'click', this.addItem);
    this.listen(this.ui.whereDone, 'click', this.showWhereDone);
    this.listen(this.ui.sortText, 'click', this.showSortText);
    this.renderList();
  }
  addItem = () => this.todos.add({ id: this.nextId++, text: this.ui.text.value || '?', done: this.ui.done.checked });
  showWhereDone = () => this.log('where(done): ' + this.todos.where({ done: true }).map(t => t.get('text')));
  showSortText = () => this.log('sort copy: ' + this.todos.sort((a,b) => a.get('text') < b.get('text') ? -1 : 1).map(t => t.get('text')));
  // renderList is internal logic → a normal method
  renderList = () => { this.ui.list.replaceChildren(...this.todos.map(t => { const li = document.createElement('li'); li.textContent = t.get('text') + (t.get('done')?' ✓':''); return li; })); };
}

new Demo().mount(app);
