Router · hash routing + Region

Routes resolve to region.show(view), so screens animate in/out. Uses native hashchange — no polling. Try the links, the back button, or a param route.

Code

import { Router, Region, View, slideIn, slideOut } from '../../src/index.js';

class Screen extends View {
  static template = '#screen';
  onMount() {
    this.ui.root.style.background = this.props.color;
    this.ui.title.textContent = this.props.title;
    this.ui.body.textContent  = this.props.body;
  }
  animateIn()  { return slideIn(this.el); }
  animateOut() { return slideOut(this.el); }
}

const main = new Region('#outlet');

new Router()
  .add('/',          () => main.show(new Screen({ title: 'Home',  body: '#/',            color: '#2563eb' })))
  .add('/about',     () => main.show(new Screen({ title: 'About', body: '#/about',       color: '#059669' })))
  .add('/users/:id', ({ id }) => main.show(new Screen({ title: 'User ' + id, body: 'param id = ' + id, color: '#7c3aed' })))
  .notFound(() => main.show(new Screen({ title: '404', body: 'no route', color: '#dc2626' })))
  .start();
