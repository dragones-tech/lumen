animate · promise-based transitions

The leave animation finishes before the node is removed — note the box never vanishes abruptly. Respects prefers-reduced-motion.

Code

import { clone, fadeIn, slideIn, slideOut } from '../../src/index.js';

async function show() {
  const el = clone('#box');
  stage.appendChild(el);
  await slideIn(el, { duration: 300 });  // resolves when the entrance ends
}

async function hide(el) {
  await slideOut(el, { duration: 300 }); // wait for the exit...
  el.remove();                            // ...THEN remove from the DOM
}
