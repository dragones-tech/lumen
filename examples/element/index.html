defineElement · a View as <counter-widget>

A normal Lumen View, registered as a Custom Element, used as a tag in plain HTML. Each tag is an independent instance; the start attribute becomes a prop.

Code

import { View, defineElement } from '../../src/index.js';

class Counter extends View {
  static template = '#counter';
  onCreate() { this.n = Number(this.props.start || 0); }
  onMount() {
    this.ui.label.textContent = this.props.label || 'count';
    this.paint();
    this.listen(this.ui.inc, 'click', () => { this.n++; this.paint(); });
    this.listen(this.ui.dec, 'click', () => { this.n--; this.paint(); });
  }
  paint = () => { this.ui.value.textContent = this.n; };
}

defineElement('counter-widget', Counter);   // now <counter-widget> works in HTML
