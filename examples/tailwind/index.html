Lumen + Tailwind

Lumen ships no CSS. Write Tailwind utilities directly in your <template> — views are Light DOM, so the global utilities just apply to the cloned nodes. Nothing to configure: Lumen handles behavior & lifecycle, Tailwind handles the look.

Code

import { View, slideIn, slideOut } from '../../src/index.js';

class Card extends View {
  static template = '#card';              // a <template> full of Tailwind classes
  onMount() {
    this.ui.title.textContent = this.props.title;
    this.listen(this.ui.remove, 'click', this.remove);
  }
  remove = () => this.props.onRemove(this);
  animateIn()  { return slideIn(this.el); }
  animateOut() { return slideOut(this.el); }
}

class Board extends View {
  static template = '#board';
  onMount() {
    this.listen(this.ui.form, 'submit', this.add);
    this.addCard('Hello');
  }
  add = (e) => {
    e.preventDefault();
    this.addCard(this.ui.input.value.trim() || 'Untitled');
    this.ui.input.value = ''; this.ui.input.focus();
  };
  addCard(title) {                         // internal logic → a method
    this.addChild(new Card({ title, onRemove: (c) => this.removeChild(c) }), this.ui.list);
  }
}

new Board().mount(document.querySelector('#app'));
