Model · one source, surgical updates

One Profile model is observed by two independent views. Each reacts only to the key it cares about — no full re-render. Type in the editor and watch the badge.

Code

import { Model, View } from '../../src/index.js';

// Handlers here use auto-bound arrow fields — one of three valid
// styles (see docs/view.md). Internal logic stays a method.
class Editor extends View {
  static template = '#editor';
  onMount() {
    const { model } = this.props;
    this.ui.name.value  = model.get('name');
    this.ui.color.value = model.get('color');
    this.listen(this.ui.name,  'input', this.onNameInput);
    this.listen(this.ui.color, 'input', this.onColorInput);
  }
  onNameInput  = () => this.props.model.set('name',  this.ui.name.value);
  onColorInput = () => this.props.model.set('color', this.ui.color.value);
}

class Badge extends View {
  static template = '#badge';
  onMount() {
    const { model } = this.props;
    this.paint();
    model.on('change', this.paint, { signal: this.signal }); // surgical, auto-cleaned
  }
  paint = () => {
    const { model } = this.props;
    this.ui.badge.textContent  = model.get('name');
    this.ui.badge.style.color  = model.get('color');
    this.ui.badge.style.borderColor = model.get('color');
  };
}

const profile = new Model({ name: 'Ada', color: '#2563eb' });
new Editor({ model: profile }).mount(app);
new Badge({ model: profile }).mount(app);
