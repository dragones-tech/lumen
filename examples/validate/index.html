Validation · one source of truth

Rules live on the Signup model. The form calls model.validate() and renders the returned errors — the model is the authority, the view just paints.

Code

import { Model, View, required, email, minLength, match } from '../../src/index.js';

class Signup extends Model {
  static rules = {                       // the single source of truth
    email:    [required(), email()],
    password: [required(), minLength(8)],
    confirm:  [match('password', 'passwords must match')],
  };
}

class SignupForm extends View {
  static template = '#signup';
  onMount() { this.listen(this.ui.form, 'submit', this.submit); }
  submit = (e) => {
    e.preventDefault();
    this.props.model.set({
      email: this.ui.email.value,
      password: this.ui.password.value,
      confirm: this.ui.confirm.value,
    });
    const errors = this.props.model.validate();   // ask the model
    this.showErrors(errors);                          // view just paints
  };
  showErrors(errors) {
    for (const f of ['email', 'password', 'confirm']) {
      this.ui[f + 'Err'].textContent = errors[f]?.[0] ?? '';
      this.ui[f].classList.toggle('bad', !!errors[f]);
    }
  }
}
